// ===== HELPERS =====
function _hash(pwd) {
  const raw = APP_SALT + ':' + pwd;
  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256, raw, Utilities.Charset.UTF_8
  );
  return bytes.map(b => ('0' + (b & 0xff).toString(16)).slice(-2)).join('').toUpperCase();
}

function _crearHoja(ss, nombre, headers) {
  let sh = ss.getSheetByName(nombre);
  if (sh) ss.deleteSheet(sh);
  sh = ss.insertSheet(nombre);
  sh.getRange(1, 1, 1, headers.length).setValues([headers])
    .setFontWeight('bold').setBackground('#E8EAED');
  sh.setFrozenRows(1);
  return sh;
}

// ===== INICIALIZADOR =====
function inicializar() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  _crearHoja(ss,'REGISTROS',[
    'id','fecha_creacion','municipio','localidad','nombre','organizacion',
    'perfil','tipo_vinculacion','que_aporta','prioridad','motivo_prioridad',
    'proximo_paso','fecha_seguimiento','estatus','observaciones',
    'creado_por','actualizado_en'
  ]);
  _crearHoja(ss,'MUNICIPIOS',['id','nombre','clave']);
  _crearHoja(ss,'LOCALIDADES',['id','municipio_id','nombre','clave']);
  _crearHoja(ss,'USUARIOS',['id','usuario','password_hash','nombre','rol','activo','created_at']);
  _crearHoja(ss,'CATALOGOS',['tipo','valor','orden']);
  _crearHoja(ss,'SEGUIMIENTOS',[
    'id','registro_id','fecha','estatus_anterior','estatus_nuevo',
    'proximo_paso','observaciones','usuario'
  ]);

  ss.getSheetByName('MUNICIPIOS')
    .getRange(2, 1, MUNICIPIOS.length, 3).setValues(MUNICIPIOS);

  const catRows = [];
  for (const tipo in CATALOGOS) {
    CATALOGOS[tipo].forEach((v, i) => catRows.push([tipo, v, i + 1]));
  }
  ss.getSheetByName('CATALOGOS')
    .getRange(2, 1, catRows.length, 3).setValues(catRows);

  ss.getSheetByName('USUARIOS').appendRow([
    1, ADMIN_USERNAME, _hash(ADMIN_PASSWORD),
    ADMIN_NAME, 'admin', true, new Date()
  ]);

  // Verificación final: leer nombres de hojas
  const nombres = ss.getSheets().map(s => s.getName());
  Logger.log('Hojas después de inicializar (' + nombres.length + '): ' + JSON.stringify(nombres));

  SpreadsheetApp.getUi().alert(
    '✓ Inicialización completa',
    'Hojas: ' + nombres.length + ' → ' + nombres.join(', ') +
    '\nMunicipios: ' + MUNICIPIOS.length +
    '\nCatálogos: ' + catRows.length + ' valores' +
    '\nUsuario: ' + ADMIN_USERNAME +
    '\n\nSIGUIENTE: crear hoja "_LOCALIDADES_SEED" y ejecutar importarLocalidades()',
    SpreadsheetApp.getUi().ButtonSet.OK
  );
}

// ===== IMPORTAR LOCALIDADES =====
function importarLocalidades() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const tmp = ss.getSheetByName('_LOCALIDADES_SEED');
  if (!tmp) throw new Error(
    'No existe la hoja "_LOCALIDADES_SEED".\n' +
    'Créala con 3 columnas (A: NOM_MUN, B: LOC, C: NOM_LOC) y pega los datos.'
  );

  const munMap = {};
  MUNICIPIOS.forEach(m => munMap[String(m[1]).trim()] = m[0]);

  const data = tmp.getDataRange().getValues();
  if (data.length < 2) throw new Error('La hoja temporal está vacía.');
  const header = data[0].map(h => String(h).trim().toUpperCase());
  if (header[0] !== 'NOM_MUN' || header[1] !== 'LOC' || header[2] !== 'NOM_LOC') {
    throw new Error('Headers incorrectos. Deben ser: NOM_MUN | LOC | NOM_LOC');
  }

  const rows = [];
  let skipped = 0;
  for (let i = 1; i < data.length; i++) {
    const [nomMun, loc, nomLoc] = data[i];
    // Solo NOM_LOC es obligatorio; NOM_MUN y LOC pueden venir vacíos
    if (!String(nomLoc).trim()) { skipped++; continue; }
    // NOM_MUN vacío = Tulancingo de Bravo (municipio único del clon)
    const nombreMun = String(nomMun).trim() || 'Tulancingo de Bravo';
    const munId = munMap[nombreMun];
    if (!munId) { skipped++; continue; }
    // LOC vacío = clave secuencial automática (0001, 0002, ...)
    // ORDEN CORRECTO: id | municipio_id | nombre | clave
    const clave = String(loc).trim() || String(rows.length + 1).padStart(4, '0');
    rows.push([rows.length + 1, munId, String(nomLoc).trim(), clave]);
  }

  if (rows.length === 0) throw new Error(
    'No se insertaron filas. Verifica que los nombres de municipio coincidan con MUNICIPIOS.'
  );

  ss.getSheetByName('LOCALIDADES')
    .getRange(2, 1, rows.length, 4).setValues(rows);

  SpreadsheetApp.getUi().alert(
    '✓ Localidades importadas',
    'Insertadas: ' + rows.length + '\nOmitidas: ' + skipped +
    '\n\nYa puedes borrar la hoja "_LOCALIDADES_SEED".',
    SpreadsheetApp.getUi().ButtonSet.OK
  );
}

// FIX (2026-09-02): columnas nombre/clave intercambiadas en LOCALIDADES.
// La primera importación escribió [id, municipio_id, CLAVE, NOMBRE] con
// headers id|municipio_id|nombre|clave → el dropdown mostraba números.
// Esta función intercambia columnas C y D (solo si C parece clave numérica,
// para que sea seguro correrla dos veces). NO toca encabezados ni ids.
// EJECUTAR UNA VEZ desde el editor.
// =========================================================
function repararLocalidades() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName('LOCALIDADES');
  if (!sh) throw new Error('No existe la hoja LOCALIDADES');
  const last = sh.getLastRow();
  if (last < 2) { SpreadsheetApp.getUi().alert('LOCALIDADES está vacía.'); return; }

  const cols = sh.getRange(2, 3, last - 1, 2).getValues(); // C y D
  let yaBien = 0, cambiadas = 0;
  for (let i = 0; i < cols.length; i++) {
    const c = String(cols[i][0]).trim();
    const d = String(cols[i][1]).trim();
    // Si C ya es texto de nombre (no numérico de 1-4 dígitos), nada que hacer
    if (!/^\d{1,4}$/.test(c)) { yaBien++; continue; }
    cols[i] = [d, c];
    cambiadas++;
  }
  if (cambiadas > 0) sh.getRange(2, 3, last - 1, 2).setValues(cols);
  SpreadsheetApp.getUi().alert(
    '✓ Localidades reparadas',
    'Intercambiadas: ' + cambiadas + '\nYa correctas: ' + yaBien,
    SpreadsheetApp.getUi().ButtonSet.OK
  );
}

// Crea el usuario "campo" con rol operador. Idempotente: si ya existe,
// no duplica. Edita las 3 constantes de abajo si quieres otro nombre.
// EJECUTAR UNA VEZ desde el editor.
// =========================================================
function agregarOperador() {
  const NUEVO_USUARIO  = 'campo';
  const NUEVO_PASSWORD = 'Campo2026';
  const NUEVO_NOMBRE   = 'Operador de Campo';

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName('USUARIOS');
  if (!sh) throw new Error('No existe la hoja USUARIOS');

  const data = sh.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][1]).trim().toLowerCase() === NUEVO_USUARIO.toLowerCase()) {
      SpreadsheetApp.getUi().alert(
        'Ya existe',
        'El usuario "' + NUEVO_USUARIO + '" ya está dado de alta (fila ' + (i + 1) + ').',
        SpreadsheetApp.getUi().ButtonSet.OK
      );
      return;
    }
  }

  sh.appendRow([
    sh.getLastRow(), NUEVO_USUARIO, _hash(NUEVO_PASSWORD),
    NUEVO_NOMBRE, 'operador', true, new Date()
  ]);
  SpreadsheetApp.getUi().alert(
    '✓ Operador creado',
    'Usuario: ' + NUEVO_USUARIO + '\nContraseña: ' + NUEVO_PASSWORD +
    '\n\nComunícala al equipo de campo y cámbiala después si hace falta.',
    SpreadsheetApp.getUi().ButtonSet.OK
  );
}
